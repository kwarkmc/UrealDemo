cmake_minimum_required(VERSION 3.16)
project(UAVSimCore LANGUAGES CXX)

enable_testing()
foreach(check IN ITEMS core_checks fixed_wing_checks navigation_checks ground_checks terrain_surface_checks systems_checks operations_checks stl_mesh_checks gimbal_checks)
    add_executable(${check} tests/${check}.cpp)
    target_include_directories(${check} PRIVATE Source/FlightCore)
    target_compile_features(${check} PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(${check} PRIVATE /W4 /permissive- /fp:precise /EHsc)
    else()
        target_compile_options(${check} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME ${check} COMMAND ${check})
endforeach()
