cmake_minimum_required(VERSION 3.16)
project(UAVSimCore LANGUAGES CXX)

enable_testing()
foreach(check IN ITEMS core_checks fixed_wing_checks navigation_checks ground_checks terrain_surface_checks systems_checks operations_checks stl_mesh_checks gimbal_checks research_checks return_checks refinement_checks propeller_visual_checks retractable_checks navigation_physics_checks)
    add_executable(${check} tests/${check}.cpp)
    target_include_directories(${check} PRIVATE Source/FlightCore)
    target_compile_features(${check} PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(${check} PRIVATE /W4 /permissive- /fp:precise /EHsc)
    else()
        target_compile_options(${check} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME ${check} COMMAND ${check})
endforeach()
target_compile_definitions(research_checks PRIVATE RESEARCH_PROFILE="${CMAKE_SOURCE_DIR}/aircraft/c208b-research.json")

target_compile_definitions(return_checks PRIVATE RESEARCH_PROFILE="${CMAKE_SOURCE_DIR}/aircraft/c208b-research.json")
target_compile_definitions(refinement_checks PRIVATE RESEARCH_PROFILE="${CMAKE_SOURCE_DIR}/aircraft/c208b-research.json")
target_compile_definitions(retractable_checks PRIVATE RESEARCH_PROFILE="${CMAKE_SOURCE_DIR}/aircraft/c208b-retractable-research.json")
target_compile_definitions(navigation_physics_checks PRIVATE RESEARCH_PROFILE="${CMAKE_SOURCE_DIR}/aircraft/c208b-retractable-research.json")
add_executable(navigation_physics_probe tests/navigation_physics_probe.cpp)
target_include_directories(navigation_physics_probe PRIVATE Source/FlightCore)
target_compile_features(navigation_physics_probe PRIVATE cxx_std_17)
target_compile_definitions(navigation_physics_probe PRIVATE RESEARCH_PROFILE="${CMAKE_SOURCE_DIR}/aircraft/c208b-research.json")
